简介

Python 是一种简洁、易读、功能强大的高级编程语言。它被广泛应用于 Web 开发、数据分析、人工智能、自动化脚本等领域。本文将介绍 Python 的基础知识。

Python 的特点

  • 简洁易读：语法简洁，接近自然语言
  • 动态类型：变量不需要声明类型
  • 丰富的库：拥有大量第三方库
  • 跨平台：可在 Windows、Mac、Linux 上运行
  • 面向对象：支持面向对象编程

第一个 Python 程序

Hello World

print("Hello, World!")

就这么简单！Python 不需要编译，直接运行即可。

基本语法

1. 变量和数据类型

Python 是动态类型语言，变量不需要声明类型：

基本数据类型：

integer = 42          # 整数
float_num = 3.14      # 浮点数
complex_num = 1 + 2j  # 复数

# 字符串

multiline = """多行
字符串"""

# 布尔类型
is_true = True
is_false = False

# 空值

数据结构：

# 列表（可变）
fruits = ["apple", "banana", "orange"]

# 字典

    "name": "张三",

2. 运算符

# 算术运算符
sum = 5 + 3       # 加
diff = 5 - 3      # 减
product = 5 * 3   # 乘
quotient = 5 / 3  # 除（结果为浮点数）
floor_div = 5 // 3  # 整除
remainder = 5 % 3   # 取模
power = 5 ** 3      # 幂运算

# 比较运算符
is_equal = (5 == 3)    # False
is_greater = (5 > 3)   # True
is_not_equal = (5 != 3)  # True

# 逻辑运算符
and_result = True and False  # False
or_result = True or False    # True
not_result = not True        # False

3. 控制流程

if-elif-else 语句：

score = 85

if score >= 90:
    print("优秀")
elif score >= 80:
    print("良好")
elif score >= 60:
    print("及格")
else:
    print("不及格")

for 循环：

# 遍历列表
fruits = ["apple", "banana", "orange"]
for fruit in fruits:
    print(fruit)

# 遍历数字范围
for i in range(5):  # 0 到 4
    print(i)

for i in range(1, 6):  # 1 到 5
    print(i)

for i in range(0, 10, 2):  # 0, 2, 4, 6, 8
    print(i)

while 循环：

count = 0
while count < 5:
    print(count)
    count += 1

numbers = [1, 2, 3, 4, 5]

函数

定义和调用函数

# 基本函数
def greet(name):
    print(f"Hello, {name}!")

greet("张三")

# 有返回值的函数
def add(a, b):
    return a + b

result = add(5, 3)

# 默认参数
def greet(name, greeting="Hello"):
    print(f"{greeting}, {name}!")

greet("张三")  # 使用默认值
greet("李四", "Hi")  # 使用自定义值

# 可变参数
def sum_all(*numbers):
    total = 0
    for num in numbers:
        total += num
    return total

result = sum_all(1, 2, 3, 4, 5)

# 关键字参数
def create_person(**kwargs):
    return kwargs

person = create_person(name="张三", age=25, city="北京")

Lambda 函数

# Lambda 表达式（匿名函数）
square = lambda x: x ** 2
print(square(5))  # 25

# 常用于 map、filter 等函数
numbers = [1, 2, 3, 4, 5]
squared = list(map(lambda x: x ** 2, numbers))
evens = list(filter(lambda x: x % 2 == 0, numbers))

列表推导式

# 基本列表推导式
squares = [x ** 2 for x in range(10)]

# 带条件的列表推导式
evens = [x for x in range(10) if x % 2 == 0]

# 字典推导式
square_dict = {x: x ** 2 for x in range(5)}

# 集合推导式
unique_squares = {x ** 2 for x in [1, -1, 2, -2, 3, -3]}

文件操作

# 写文件
with open("example.txt", "w", encoding="utf-8") as f:
    f.write("Hello, World!\n")
    f.write("Python is awesome!")

# 读文件
with open("example.txt", "r", encoding="utf-8") as f:
    content = f.read()  # 读取全部内容
    print(content)

# 按行读取
with open("example.txt", "r", encoding="utf-8") as f:
    for line in f:
        print(line.strip())

# 追加内容
with open("example.txt", "a", encoding="utf-8") as f:
    f.write("\nNew line")

异常处理

try:
    result = 10 / 0
except ZeroDivisionError:
    print("除数不能为零！")
except Exception as e:
    print(f"发生错误: {e}")
else:
    print("没有发生异常")
finally:
    print("清理资源")

类和对象

class Person:
    # 构造方法
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    # 实例方法
    def introduce(self):
        print(f"我是 {self.name}，今年 {self.age} 岁。")
    
    # 类方法
    @classmethod
    def from_birth_year(cls, name, birth_year):
        age = 2025 - birth_year
        return cls(name, age)
    
    # 静态方法
    @staticmethod
    def is_adult(age):
        return age >= 18

# 创建对象
person1 = Person("张三", 25)
person1.introduce()

person2 = Person.from_birth_year("李四", 2000)
print(Person.is_adult(25))  # True
